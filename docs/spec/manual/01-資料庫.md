# 資料庫設計

下列 SQL 即為第一版遷移檔內容（`db/migrations/001_init.sql` 起），共 24 張表。狀態類欄位用 `text` 加 CHECK，方便日後增加選項。

## 1. 帳號與權限

```sql
CREATE EXTENSION IF NOT EXISTS pgcrypto;
CREATE EXTENSION IF NOT EXISTS timescaledb;

CREATE TABLE roles (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name        text NOT NULL UNIQUE,
  description text,
  perms       text[] NOT NULL DEFAULT '{}',   -- 權限碼，見後端邏輯 §7
  is_system   boolean NOT NULL DEFAULT false, -- 系統管理員不可刪
  created_at  timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE users (
  id             uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  username       text NOT NULL UNIQUE,
  display_name   text NOT NULL,
  password_hash  text NOT NULL,              -- argon2id
  role_id        uuid NOT NULL REFERENCES roles(id),
  phone          text,
  email          text,
  enabled        boolean NOT NULL DEFAULT true,
  failed_logins  int NOT NULL DEFAULT 0,
  locked_until   timestamptz,
  last_login_at  timestamptz,
  created_at     timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE refresh_tokens (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id     uuid NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  token_hash  text NOT NULL,
  expires_at  timestamptz NOT NULL,
  revoked_at  timestamptz
);

CREATE TABLE op_logs (
  id         bigserial PRIMARY KEY,
  at         timestamptz NOT NULL DEFAULT now(),
  user_id    uuid REFERENCES users(id),
  username   text NOT NULL,                  -- 冗餘保存，帳號刪除後仍可查
  action     text NOT NULL,                  -- e.g. robot.estop, task.approve, settings.update
  target     text,                           -- e.g. ANBOT-003, task:uuid
  detail     jsonb,
  ip         inet
);
CREATE INDEX ON op_logs (at DESC);
CREATE INDEX ON op_logs (user_id, at DESC);
```

## 2. 場域與設備

```sql
CREATE TABLE floors (
  id          text PRIMARY KEY,               -- '1f' | '2f' | 'b1'
  name        text NOT NULL,                  -- '1F 辦公室'
  sort        int NOT NULL,
  plan_key    text,                           -- MinIO 中的平面圖
  width_m     numeric(8,2),                   -- 平面圖實際寬度
  height_m    numeric(8,2),
  robot_quota int NOT NULL DEFAULT 0          -- 該樓層配置台數
);

CREATE TABLE robots (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  code         text NOT NULL UNIQUE,          -- ANBOT-001
  brand        text NOT NULL CHECK (brand IN ('anbot','unitree','deep')),
  model        text NOT NULL,                 -- ANBOT-Y, Go2-X ...
  serial_no    text,
  floor_id     text NOT NULL REFERENCES floors(id),
  adapter      text NOT NULL,                 -- 對應哪個轉接器實例
  conn         jsonb NOT NULL DEFAULT '{}',   -- IP、埠號等，敏感欄位加密
  caps         text[] NOT NULL DEFAULT '{}',  -- ptz, light, talk, follow, lift, env
  modules      jsonb NOT NULL DEFAULT '{}',
  params       jsonb NOT NULL DEFAULT '{}',   -- 四個子系統參數
  enabled      boolean NOT NULL DEFAULT true,
  created_at   timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE robot_map_calib (
  robot_id     uuid PRIMARY KEY REFERENCES robots(id) ON DELETE CASCADE,
  robot_map_id text NOT NULL,                 -- 機器人端地圖名稱
  matrix       double precision[6] NOT NULL,  -- 仿射 [a,b,tx,c,d,ty]，機器人→平台
  points       jsonb NOT NULL,                -- 校正用的對應點
  rmse_m       numeric(6,3) NOT NULL,
  updated_at   timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE chargers (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  code        text NOT NULL UNIQUE,           -- CHG-1F-1
  floor_id    text NOT NULL REFERENCES floors(id),
  x           double precision NOT NULL,
  y           double precision NOT NULL,
  brand       text NOT NULL,                  -- 只供該品牌使用
  robot_dock  jsonb NOT NULL DEFAULT '{}',    -- 機器人端的充電樁 ID
  occupied_by uuid REFERENCES robots(id),
  enabled     boolean NOT NULL DEFAULT true
);

CREATE TABLE cameras (
  id         uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  code       text NOT NULL UNIQUE,            -- CAM-1F-07
  name       text NOT NULL,
  floor_id   text NOT NULL REFERENCES floors(id),
  x          double precision NOT NULL,
  y          double precision NOT NULL,
  rtsp_url   bytea NOT NULL,                  -- pgp_sym_encrypt
  stream_key text NOT NULL UNIQUE,            -- MediaMTX 路徑名
  enabled    boolean NOT NULL DEFAULT true
);
```

## 3. 導航與任務

```sql
CREATE TABLE pois (                           -- 興趣點
  id       uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  floor_id text NOT NULL REFERENCES floors(id),
  name     text NOT NULL,
  x double precision NOT NULL, y double precision NOT NULL, yaw double precision
);

CREATE TABLE zones (                          -- 特殊區域
  id       uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  floor_id text NOT NULL REFERENCES floors(id),
  name     text NOT NULL,
  kind     text NOT NULL CHECK (kind IN ('forbidden','slow','silent')),
  polygon  jsonb NOT NULL                     -- [[x,y],...]
);

CREATE TABLE paths (
  id       uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  floor_id text NOT NULL REFERENCES floors(id),
  name     text NOT NULL,
  points   jsonb NOT NULL,                    -- [{x,y,yaw?,dwellSec?,action?}]
  length_m numeric(8,1) NOT NULL
);

CREATE TABLE patrol_schedules (
  id        uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name      text NOT NULL,
  path_id   uuid NOT NULL REFERENCES paths(id),
  robot_id  uuid REFERENCES robots(id),       -- null = 由派工選車
  start_time time NOT NULL,
  repeat    text NOT NULL CHECK (repeat IN ('daily','weekdays','weekends','once')),
  once_date date,
  loops     int NOT NULL DEFAULT 1,
  priority  text NOT NULL DEFAULT 'P3',
  enabled   boolean NOT NULL DEFAULT true
);

CREATE TABLE task_templates (
  id       uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name     text NOT NULL,
  kind     text NOT NULL,                     -- inspect, env, security, equipment, gas ...
  priority text NOT NULL,
  floor_id text REFERENCES floors(id),
  target   jsonb                              -- {poiId} | {pathId}
);

CREATE TABLE tasks (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  code          text NOT NULL UNIQUE,         -- T-20261011-0042
  kind          text NOT NULL,
  priority      text NOT NULL CHECK (priority IN ('P1','P2','P3','P4')),
  source        text NOT NULL CHECK (source IN ('manual','schedule','alarm','api')),
  floor_id      text NOT NULL REFERENCES floors(id),
  target        jsonb NOT NULL,               -- {poiId|pathId|x,y}
  robot_id      uuid REFERENCES robots(id),
  pinned_robot  boolean NOT NULL DEFAULT false,
  status        text NOT NULL CHECK (status IN
                  ('pending_approval','queued','assigned','running','paused',
                   'done','cancelled','failed','rejected')),
  created_by    uuid REFERENCES users(id),
  approved_by   uuid REFERENCES users(id),
  approved_at   timestamptz,
  reject_reason text,
  preempted_by  uuid REFERENCES tasks(id),    -- 被哪個 P1 中斷
  progress      jsonb NOT NULL DEFAULT '{}',  -- {wpIndex, loop}
  note          text,
  created_at    timestamptz NOT NULL DEFAULT now(),
  started_at    timestamptz,
  ended_at      timestamptz,
  fail_reason   text
);
CREATE INDEX ON tasks (status, floor_id);
CREATE INDEX ON tasks (robot_id, created_at DESC);
CREATE INDEX ON tasks (created_at DESC);

CREATE TABLE task_events (                    -- 任務歷程
  id      bigserial PRIMARY KEY,
  task_id uuid NOT NULL REFERENCES tasks(id) ON DELETE CASCADE,
  at      timestamptz NOT NULL DEFAULT now(),
  type    text NOT NULL,                      -- created, approved, assigned, started, wp_reached, paused, resumed, done, failed
  data    jsonb
);
CREATE INDEX ON task_events (task_id, at);
```

## 4. 告警

```sql
CREATE TABLE alarm_rules (                    -- 告警設置
  type          text PRIMARY KEY,             -- battery_low, offline, intrusion, blacklist ...
  category      text NOT NULL CHECK (category IN ('system','business')),
  name          text NOT NULL,
  level         int NOT NULL CHECK (level IN (1,2,3)),
  popup         boolean NOT NULL,
  sound         boolean NOT NULL,
  need_handle   boolean NOT NULL,
  sms_user_ids  uuid[] NOT NULL DEFAULT '{}',
  dedup_sec     int NOT NULL DEFAULT 60       -- 同來源同類型合併窗口
);

CREATE TABLE alarms (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  code         text NOT NULL UNIQUE,          -- A-20261011-0007
  type         text NOT NULL REFERENCES alarm_rules(type),
  category     text NOT NULL,
  level        int NOT NULL,
  floor_id     text REFERENCES floors(id),
  robot_id     uuid REFERENCES robots(id),
  camera_id    uuid REFERENCES cameras(id),
  x double precision, y double precision,
  message      text NOT NULL,
  confidence   numeric(5,2),                  -- AI 信心度 0–100
  status       text NOT NULL CHECK (status IN
                 ('review','open','handling','reported','closed','false_alarm','info')),
  handler_id   uuid REFERENCES users(id),
  snapshot_key text,
  clip_key     text,
  ext          jsonb,                         -- 人臉事件原始資料等
  repeat_count int NOT NULL DEFAULT 1,
  occurred_at  timestamptz NOT NULL,
  closed_at    timestamptz
);
CREATE INDEX ON alarms (status, level);
CREATE INDEX ON alarms (occurred_at DESC);
CREATE INDEX ON alarms (robot_id, occurred_at DESC);

CREATE TABLE alarm_events (                   -- 處理歷程
  id       bigserial PRIMARY KEY,
  alarm_id uuid NOT NULL REFERENCES alarms(id) ON DELETE CASCADE,
  at       timestamptz NOT NULL DEFAULT now(),
  user_id  uuid REFERENCES users(id),
  from_status text, to_status text,
  note     text
);
```

## 5. 時序與媒體

```sql
CREATE TABLE telemetry (
  at        timestamptz NOT NULL,
  robot_id  uuid NOT NULL,
  x double precision, y double precision, yaw double precision,
  battery   real,
  charging  boolean,
  state     text,
  task_id   uuid,
  rssi      smallint,
  speed     real
);
SELECT create_hypertable('telemetry', 'at', chunk_time_interval => interval '1 day');
CREATE INDEX ON telemetry (robot_id, at DESC);
ALTER TABLE telemetry SET (timescaledb.compress, timescaledb.compress_segmentby = 'robot_id');
SELECT add_compression_policy('telemetry', interval '7 days');
SELECT add_retention_policy('telemetry', interval '90 days');

CREATE TABLE robot_state_log (                -- 狀態變化（運行日誌、可用率）
  at       timestamptz NOT NULL,
  robot_id uuid NOT NULL,
  state    text NOT NULL,
  reason   text
);
SELECT create_hypertable('robot_state_log', 'at');

CREATE TABLE env_readings (                   -- 第二期
  at timestamptz NOT NULL, robot_id uuid NOT NULL, floor_id text,
  x double precision, y double precision,
  kind text NOT NULL,                         -- temp, humidity, co, smoke ...
  value real NOT NULL, unit text NOT NULL
);
SELECT create_hypertable('env_readings', 'at');

CREATE TABLE media (
  id         uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  kind       text NOT NULL CHECK (kind IN ('snapshot','clip','audio','image')),
  robot_id   uuid REFERENCES robots(id),
  camera_id  uuid REFERENCES cameras(id),
  alarm_id   uuid REFERENCES alarms(id),
  task_id    uuid REFERENCES tasks(id),
  object_key text NOT NULL,
  size_bytes bigint,
  duration_s int,
  created_by uuid REFERENCES users(id),
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ON media (created_at DESC);
```

## 7. 資料保存

| 資料 | 保存 | 做法 |
| --- | --- | --- |
| telemetry | 90 天 | TimescaleDB retention policy；7 天後壓縮 |
| robot\_state\_log、env\_readings | 1 年 | retention policy |
| tasks、alarms、op\_logs | 1 年 | 每日排程刪除過期資料並先匯出 CSV 備查 |
| media（MinIO） | 30 天 | MinIO lifecycle 規則；告警關聯檔案保存 1 年 |

以上天數全部可設定，上線前依硬碟容量與公司政策確認。
